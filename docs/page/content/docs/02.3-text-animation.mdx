# Text

Use `tween` to replace a string, `retype` to type it, or `value` to count numbers.

## Replacing text

Playing a new `text` fades out the old whole string, then fades in the new one. The text handle and styling stay the same.

```lua live
local scene = vis.scene {}

local line = scene:text {text = "VISCRIPT", size = 42, fill = "accent"}
scene:tween({target = line, text = "Hello, API"}, {duration = 1.4})

return scene
```

```js live
const scene = vis.scene();

const line = scene.text({text: "VISCRIPT", size: 42, fill: "accent"});
scene.tween({target: line, text: "Hello, API"}, {duration: 1.4});

return scene;
```

## Typing

Use `retype = true` / `retype: true` to erase back to the shared prefix (`Hello, `), then type the rest. Set it when typing from an empty string too.

```lua live
local scene = vis.scene {}

local line = scene:text {text = "", point = {-4, 0}, align = {0, 0.5}, size = 40}

scene:tween({target = line, text = "Hello, viscript", retype = true}, {duration = 1.2, curve = "linear"})
scene:wait(0.5)
scene:tween({target = line, text = "Hello, world", retype = true}, {duration = 1, curve = "linear"})
scene:wait(0.8)

return scene
```

```js live
const scene = vis.scene();

const line = scene.text({text: "", point: [-4, 0], align: [0, 0.5], size: 40});

scene.tween({target: line, text: "Hello, viscript", retype: true}, {duration: 1.2, curve: "linear"});
scene.wait(0.5);
scene.tween({target: line, text: "Hello, world", retype: true}, {duration: 1, curve: "linear"});
scene.wait(0.8);

return scene;
```

## Character sweep

Use `sweep = true` / `sweep: true` for a stylized character sweep that temporarily mixes both strings. `sweep` and `retype` require `text` and cannot both be true.

```lua live
local scene = vis.scene {}

local line = scene:text {text = "LOADING", size = 42, font = "JetBrainsMono", fill = "accent"}
scene:tween({target = line, text = "SUCCESS", sweep = true}, {duration = 1.4, curve = "linear"})

return scene
```

```js live
const scene = vis.scene();

const line = scene.text({text: "LOADING", size: 42, font: "JetBrainsMono", fill: "accent"});
scene.tween({target: line, text: "SUCCESS", sweep: true}, {duration: 1.4, curve: "linear"});

```

## Counting numbers

`number` is a text that formats a `value`. Playing `value` counts through every number in between; `tabular = true` keeps the digits from jittering.

```lua live
local scene = vis.scene {}

local n = scene:number {value = 0, point = {0, 0}, align = {0.5, 0.5}, size = 72,
                        decimals = 1, suffix = " %", tabular = true}

scene:tween({target = n, value = 99.9}, {duration = 2, curve = "ease_out"})
scene:wait(0.8)

return scene
```

```js live
const scene = vis.scene();

const n = scene.number({value: 0, point: [0, 0], align: [0.5, 0.5], size: 72,
                         decimals: 1, suffix: " %", tabular: true});

scene.tween({target: n, value: 99.9}, {duration: 2, curve: "ease_out"});
scene.wait(0.8);

return scene;
```

## Letter by letter

`spans` with `split = "letters"` makes one handle per letter, so any array call with a `lag` staggers them.

```lua live
local scene = vis.scene {}

local word = scene:spans {"viscript", point = {0, 0}, align = {0.5, 0.5}, size = 72, split = "letters"}

scene:write(word:children(), {duration = 0.4, curve = "smooth", lag = 0.12})
scene:wait(0.8)

return scene
```

```js live
const scene = vis.scene();

const word = scene.spans(["viscript"], {point: [0, 0], align: [0.5, 0.5], size: 72, split: "letters"});

scene.write(word.children(), {duration: 0.4, curve: "smooth", lag: 0.12});
scene.wait(0.8);

return scene;
```

## Replacing a label

`fade_transform` swaps two separate text objects in place, including their styles and layout. Create the new text right before the call.

```lua live
local scene = vis.scene {}

local step = scene:text {text = "1. Split the list", size = 40}
scene:fade_in(step, {shift = {0, -0.3}, duration = 0.5})
scene:wait(0.6)

for _, label in ipairs({"2. Sort each half", "3. Merge the halves"}) do
    local next_step = scene:text {text = label, size = 40}
    scene:fade_transform(step, next_step, {duration = 0.6})
    step = next_step
    scene:wait(0.6)
end

return scene
```

```js live
const scene = vis.scene();

let step = scene.text({text: "1. Split the list", size: 40});
scene.fadeIn(step, {shift: [0, -0.3], duration: 0.5});
scene.wait(0.6);

for (const label of ["2. Sort each half", "3. Merge the halves"]) {
    const next = scene.text({text: label, size: 40});
    scene.fadeTransform(step, next, {duration: 0.6});
    step = next;
    scene.wait(0.6);
}

return scene;
```

## Matching words

`match_transform` pairs equal pieces of two texts and slides them into their new places; the rest fades. With `split = "words"` the pieces are words, so keep tokens apart with spaces.

```lua live
local scene = vis.scene {}

local before = scene:spans {"total = price + tax", point = {0, 0}, align = {0.5, 0.5}, size = 34,
                           font = "JetBrainsMono", split = "words"}
scene:fade_in(before, {duration = 0.4})
scene:wait(0.5)

local after = scene:spans {"total = ( price + tax ) * qty", point = {0, 0}, align = {0.5, 0.5}, size = 34,
                          font = "JetBrainsMono", split = "words"}
scene:match_transform(before, after, {duration = 1.2})
scene:wait(0.8)

return scene
```

```js live
const scene = vis.scene();

const before = scene.spans(["total = price + tax"], {point: [0, 0], align: [0.5, 0.5], size: 34,
                                                    font: "JetBrainsMono", split: "words"});
scene.fadeIn(before, {duration: 0.4});
scene.wait(0.5);

const after = scene.spans(["total = ( price + tax ) * qty"], {point: [0, 0], align: [0.5, 0.5], size: 34,
                                                             font: "JetBrainsMono", split: "words"});
scene.matchTransform(before, after, {duration: 1.2});
scene.wait(0.8);

return scene;
```

## Matching formulas

`match_transform` slides the parts two formulas share into their new places and fades the rest. Create the target right before the call.

```lua live
local scene = vis.scene {}

local a = scene:tex {tex = "a^2 + b^2 = c^2", point = {0, 0}, size = 60}
scene:fade_in(a, {duration = 0.6})
scene:wait(0.4)

local b = scene:tex {tex = "c^2 = a^2 + b^2", point = {0, 0}, size = 60}
scene:match_transform(a, b, {duration = 1.4})
scene:wait(0.8)

return scene
```

```js live
const scene = vis.scene();

```

## Code: highlight and edit

`code` returns tokens you can search. `highlight` dims everything but the selection; `edit` replaces tokens and animates the diff.

```lua live
local scene = vis.scene {}

local listing = scene:code {text = "local total = 0\nfor i = 1, n do\n    total = total + i\nend\nreturn total",
                            language = "lua", point = {0, 0}, align = {0.5, 0.5}, size = 26}
scene:fade_in(listing, {duration = 0.4})
scene:wait(0.3)

scene:highlight(listing, listing:search("total", {match = "all", whole = true}))
scene:wait(0.8)
scene:highlight(listing)
scene:edit(listing, {replace = "total", with = "sum", match = "all"})
scene:wait(1)

return scene
```

```js live
const scene = vis.scene();

const listing = scene.code({text: "local total = 0\nfor i = 1, n do\n    total = total + i\nend\nreturn total",

scene.wait(0.3);

scene.highlight(listing, listing.search("total", {match: "all", whole: true}));
```
