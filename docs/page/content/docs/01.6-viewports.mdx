# Viewports

<p>A finished child scene can be attached inside a parent with <code>parent:viewport(child, &#123;x, y, width, height&#125;)</code>, in
fractions of the parent frame.<br />
Each child keeps its own camera, theme and timeline — useful for split screens, insets and comparisons.</p>
<p>Viewports suit a poster or cheat sheet: each card is a small scene with its own camera and timeline, so every card can be written, framed and animated on its own, then laid out on one page.<br />
For the same layout across themes, see <a href="#/docs/themes">Themes</a>.</p>

To play whole scenes one after another instead, see [Scene](#/docs/scene-transitions).

```lua live
local panelTheme = {preset = "pro_black", background = "#161b22"}
local root = vis.scene {width = 960, height = 540, theme = {preset = "pro_black", background = "#0d1117"},
                        camera = {height = 6}}
root:text {text = "Trigonometry at a glance", role = "h2", point = {0, 2.62}}

-- One card of the sheet: its own scene, camera and timeline, with a caption at the bottom.
local function card(caption, build)
    local s = vis.scene {width = 312, height = 222, theme = panelTheme, camera = {height = 3.6}}
    s:text {text = caption, point = {0, -1.55}, size = 13, color = "muted"}
    build(s)
    return s
end

local cards = {
    card("unit circle", function(s)
        local ring = s:circle {stroke = "muted", width = 1.5}
        local ray = s:line {from = {0, 0}, to = {math.cos(0.9), math.sin(0.9)}, stroke = "accent", width = 3}
        local mark = s:angle {vertex = {0, 0}, from = {1, 0}, to = {math.cos(0.9), math.sin(0.9)}, radius = 0.35,
                              stroke = "warning", width = 2, label = "θ"}
        s:create(ring, 0.6)
        s:create(ray, 0.4)
        s:create(mark, 0.4)
    end),
    card("sin and cos", function(s)
        local axes = s:space {x = {-2.2, 2.2, 1.1}, y = {-1.1, 1.1, 1}}
        local sin = axes:plot {fn = function(x) return math.sin(1.4 * x) end, x_range = {-2.2, 2.2, 0.05}, stroke = "accent", width = 2.5}
        local cos = axes:plot {fn = function(x) return math.cos(1.4 * x) end, x_range = {-2.2, 2.2, 0.05}, stroke = "warning", width = 2.5}
        s:create(axes, 0.4)
        s:create({sin, cos}, 1, "smooth", 0.2)
    end),
    card("Pythagorean identity", function(s)
        local eq = s:tex {tex = {{"\\sin^2\\theta", color = "accent"}, "+", {"\\cos^2\\theta", color = "warning"}, "=", "1"},
                          point = {0, 0.1}, size = 26}
        s:write(eq, 1)
    end),
    card("right triangle", function(s)
        local tri = s:polygon {points = {{-1.3, -0.9}, {1.3, -0.9}, {1.3, 0.9}}, stroke = "foreground", width = 2}
        local h = s:text {text = "hyp", point = {-0.2, 0.25}, size = 14, color = "muted"}
        local o = s:text {text = "opp", point = {1.6, 0}, size = 14, color = "accent"}
        local a = s:text {text = "adj", point = {0, -1.15}, size = 14, color = "warning"}
        s:create(tri, 0.6)
        s:write(o, 0.3)
        s:write(a, 0.3)
        s:write(h, 0.3)
    end),
    card("special angles", function(s)
        local t = s:table {header = {"θ", "sin", "cos"},
                           rows = {{"0", "0", "1"}, {"π/6", "1/2", "√3/2"}, {"π/4", "√2/2", "√2/2"}, {"π/2", "1", "0"}},
                           point = {0, 0.15}, size = 13, stripes = true}
        s:fade_in(t, {duration = 0.6})
    end),
    card("derivatives", function(s)
        local d1 = s:tex {tex = "\\frac{d}{dx}\\sin x = \\cos x", point = {0, 0.6}, size = 18}
        local d2 = s:tex {tex = "\\frac{d}{dx}\\cos x = -\\sin x", point = {0, -0.5}, size = 18}
        s:write(d1, 0.8)
        s:write(d2, 0.8)
    end),
}

-- Three columns and two rows below the title, with a gutter between cards.
local top, gutter = 0.14, 0.012
local w, h = (1 - 4 * gutter) / 3, (1 - top - 3 * gutter) / 2
for i, s in ipairs(cards) do
    local column, row = (i - 1) % 3, (i - 1) // 3
    root:viewport(s, {x = gutter + column * (w + gutter), y = top + gutter + row * (h + gutter), width = w, height = h})
end

return root
```

```js live
const panelTheme = {preset: "pro_black", background: "#161b22"};
const root = vis.scene({width: 960, height: 540, theme: {preset: "pro_black", background: "#0d1117"},
                        camera: {height: 6}});
root.text({text: "Trigonometry at a glance", role: "h2", point: [0, 2.62]});

// One card of the sheet: its own scene, camera and timeline, with a caption at the bottom.
function card(caption, build) {
    const s = vis.scene({width: 312, height: 222, theme: panelTheme, camera: {height: 3.6}});
    s.text({text: caption, point: [0, -1.55], size: 13, color: "muted"});
    build(s);
    return s;
}

const tip = [Math.cos(0.9), Math.sin(0.9)];
const cards = [
    card("unit circle", (s) => {
        const ring = s.circle({stroke: "muted", width: 1.5});
        const ray = s.line({from: [0, 0], to: tip, stroke: "accent", width: 3});
        const mark = s.angle({vertex: [0, 0], from: [1, 0], to: tip, radius: 0.35,
                              stroke: "warning", width: 2, label: "θ"});
        s.create(ring, 0.6);
        s.create(ray, 0.4);
        s.create(mark, 0.4);
    }),
    card("sin and cos", (s) => {
        const axes = s.space({x: [-2.2, 2.2, 1.1], y: [-1.1, 1.1, 1]});
        const sin = axes.plot({fn: (x) => Math.sin(1.4 * x), x_range: [-2.2, 2.2, 0.05], stroke: "accent", width: 2.5});
        const cos = axes.plot({fn: (x) => Math.cos(1.4 * x), x_range: [-2.2, 2.2, 0.05], stroke: "warning", width: 2.5});
        s.create(axes, 0.4);
        s.create([sin, cos], 1, "smooth", 0.2);
    }),
    card("Pythagorean identity", (s) => {
        const eq = s.tex({tex: [{text: "\\sin^2\\theta", color: "accent"}, "+", {text: "\\cos^2\\theta", color: "warning"}, "=", "1"],
                          point: [0, 0.1], size: 26});
        s.write(eq, 1);
    }),
    card("right triangle", (s) => {
        const tri = s.polygon({points: [[-1.3, -0.9], [1.3, -0.9], [1.3, 0.9]], stroke: "foreground", width: 2});
        const h = s.text({text: "hyp", point: [-0.2, 0.25], size: 14, color: "muted"});
        const o = s.text({text: "opp", point: [1.6, 0], size: 14, color: "accent"});
        const a = s.text({text: "adj", point: [0, -1.15], size: 14, color: "warning"});
        s.create(tri, 0.6);
        s.write(o, 0.3);
        s.write(a, 0.3);
        s.write(h, 0.3);
    }),
    card("special angles", (s) => {
        const t = s.table({header: ["θ", "sin", "cos"],
                           rows: [["0", "0", "1"], ["π/6", "1/2", "√3/2"], ["π/4", "√2/2", "√2/2"], ["π/2", "1", "0"]],
                           point: [0, 0.15], size: 13, stripes: true});
        s.fadeIn(t, {duration: 0.6});
    }),
    card("derivatives", (s) => {
        const d1 = s.tex({tex: "\\frac{d}{dx}\\sin x = \\cos x", point: [0, 0.6], size: 18});
        const d2 = s.tex({tex: "\\frac{d}{dx}\\cos x = -\\sin x", point: [0, -0.5], size: 18});
        s.write(d1, 0.8);
        s.write(d2, 0.8);
    }),
];

// Three columns and two rows below the title, with a gutter between cards.
const top = 0.14, gutter = 0.012;
const w = (1 - 4 * gutter) / 3, h = (1 - top - 3 * gutter) / 2;
cards.forEach((s, i) => {
    const column = i % 3, row = Math.floor(i / 3);
    root.viewport(s, {x: gutter + column * (w + gutter), y: top + gutter + row * (h + gutter), width: w, height: h});
});

return root;
```
