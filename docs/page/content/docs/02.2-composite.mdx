# Composite

<p><code>scene:tween(descriptor, options)</code> changes several properties of one target in a single clip; <code>scene:set</code> applies
the same descriptor instantly.</p>
<table>
<tr><th>Group</th><th>Descriptor keys</th></tr>
<tr><td>Geometry</td><td><code>shift</code> <em>or</em> <code>transform</code>, <code>arc</code>, <code>size</code>, <code>corner</code>, <code>radius</code></td></tr>
<tr><td>Paint</td><td><code>fill</code>, <code>stroke</code>, <code>width</code>, <code>opacity</code>, <code>gradient</code></td></tr>
<tr><td>Stroke</td><td><code>progress</code>, <code>dash_offset</code>, <code>tail</code>, <code>tip</code></td></tr>
<tr><td>Effects</td><td><code>blur</code>, <code>shadow</code></td></tr>
</table>

## Many properties, one clip

Every key in the descriptor moves together on the same curve, so position, size, corners and colour change as one motion.

```lua live
local scene = vis.scene {}

local card = scene:rectangle {center = {-3, 0}, size = {1.2, 1.2}, corner = 0, fill = "accent"}

-- one clip, four properties, one curve
scene:tween({target = card, shift = {6, 0}, size = {2.8, 1.4}, corner = 0.7, fill = "danger"}, {duration = 1.6, curve = "smooth"})
scene:wait(0.3)
scene:tween({target = card, shift = {-6, 0}, size = {1.2, 1.2}, corner = 0, fill = "accent"}, {duration = 1.2})

return scene
```

```js live
const scene = vis.scene();

const card = scene.rectangle({center: [-3, 0], size: [1.2, 1.2], corner: 0, fill: "accent"});

// one clip, four properties, one curve
scene.tween({target: card, shift: [6, 0], size: [2.8, 1.4], corner: 0.7, fill: "danger"}, {duration: 1.6, curve: "smooth"});
scene.wait(0.3);
scene.tween({target: card, shift: [-6, 0], size: [1.2, 1.2], corner: 0, fill: "accent"}, {duration: 1.2});

return scene;
```

## Moving on an arc

`arc` bends a `shift` into a circular arc by that angle. Without it, points travel in a straight line.

```lua live
local scene = vis.scene {}

local straight = scene:circle {center = {-3.5, -1}, radius = 0.3, fill = "accent"}
local curved = scene:circle {center = {-3.5, -1}, radius = 0.3, fill = "warning"}

scene:parallel(
    function() scene:tween({target = straight, shift = {7, 0}}, {duration = 1.6}) end,
    function() scene:tween({target = curved, shift = {7, 0}, arc = -math.pi / 2}, {duration = 1.6}) end)
scene:wait(0.5)

return scene
```

```js live
const scene = vis.scene();

const straight = scene.circle({center: [-3.5, -1], radius: 0.3, fill: "accent"});
const curved = scene.circle({center: [-3.5, -1], radius: 0.3, fill: "warning"});

scene.parallel(
    () => scene.tween({target: straight, shift: [7, 0]}, {duration: 1.6}),
    () => scene.tween({target: curved, shift: [7, 0], arc: -Math.PI / 2}, {duration: 1.6}));
scene.wait(0.5);

return scene;
```

## tween vs set

`tween` blends over its duration; `set` jumps at the cursor. A `set` before any animation defines the starting state.

```lua live
local scene = vis.scene {}

local a = scene:rectangle {center = {-2, 0}, size = {1.6, 1.6}, fill = "accent"}
local b = scene:rectangle {center = {2, 0}, size = {1.6, 1.6}, fill = "accent"}
scene:set({target = b, opacity = 0.3})   -- before any animation: b starts faded

scene:wait(0.5)
scene:tween({target = a, fill = "danger"}, {duration = 1})    -- blends over a second
scene:set({target = b, fill = "danger", opacity = 1})   -- jumps at the cursor
scene:wait(1.5)

return scene
```

```js live
const scene = vis.scene();

const a = scene.rectangle({center: [-2, 0], size: [1.6, 1.6], fill: "accent"});
const b = scene.rectangle({center: [2, 0], size: [1.6, 1.6], fill: "accent"});
scene.set({target: b, opacity: 0.3});   // before any animation: b starts faded

scene.wait(0.5);
scene.tween({target: a, fill: "danger"}, {duration: 1});    // blends over a second
scene.set({target: b, fill: "danger", opacity: 1});   // jumps at the cursor
scene.wait(1.5);

return scene;
```

<table>
<tr><th></th><th><code>tween</code></th><th><code>set</code></th></tr>
<tr><td>Timing</td><td>Interpolates over <code>duration</code></td><td>Instant</td></tr>
<tr><td>Before first animation</td><td>—</td><td>Defines the starting state</td></tr>
<tr><td><code>arc</code></td><td>Points follow a circular arc</td><td>Not allowed</td></tr>
</table>

## A list of descriptors

`tween` also takes a list, one descriptor per target, and the options table's `lag` field staggers them.

```lua live
local scene = vis.scene {}

local bars, steps = {}, {}
for i = 1, 6 do
    bars[i] = scene:rectangle {center = {-3.75 + i * 1.1, -2}, size = {0.8, 0.4}, fill = "accent"}
    steps[i] = {target = bars[i], shift = {0, i * 0.3}, size = {0.8, 0.4 + i * 0.6}, fill = "success"}
end

scene:tween(steps, {duration = 0.6, curve = "back", lag = 0.12})   -- a list of descriptors, staggered by lag
scene:wait(0.6)

return scene
```

```js live
const scene = vis.scene();

const steps = [];
for (let i = 1; i <= 6; i++) {
    const bar = scene.rectangle({center: [-3.75 + i * 1.1, -2], size: [0.8, 0.4], fill: "accent"});
    steps.push({target: bar, shift: [0, i * 0.3], size: [0.8, 0.4 + i * 0.6], fill: "success"});
}

scene.tween(steps, {duration: 0.6, curve: "back", lag: 0.12});   // a list of descriptors, staggered by lag
scene.wait(0.6);

return scene;
```
