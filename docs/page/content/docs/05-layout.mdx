# Layout & Exits

Place objects relative to each other, rearrange them, take them off stage, and connect them.

## Relative layout

<p>Layout helpers are object methods that run <strong>before the first timeline call</strong>. They read authored bounds, so you
describe intent ("to the right of", "a row of") instead of computing coordinates.</p>
<table>
<tr><th>Method</th><th>Receiver</th><th>Arguments</th></tr>
<tr><td><code>obj:move_to(point)</code></td><td>Any object</td><td>Bounds centre to a point.</td></tr>
<tr><td><code>obj:next_to(other, direction, gap)</code></td><td>Sibling</td><td><code>direction</code> is a vector such as <code>&#123;1, 0&#125;</code>.</td></tr>
<tr><td><code>obj:align_to(other, direction)</code></td><td>Sibling</td><td>Align one edge with another object's edge.</td></tr>
<tr><td><code>group:arrange(direction, gap)</code></td><td>Group</td><td>Children in a row or column.</td></tr>
<tr><td><code>group:arrange_grid(columns, col_gap, row_gap)</code></td><td>Group</td><td>Children in a grid.</td></tr>
</table>

### Rows and columns

`arrange(direction, gap)` lines a group's children up: `{1, 0}` runs left to right, `{0, -1}` top to bottom. `move_to` then places the whole group.

```lua live
local scene = vis.scene {theme = "pro_black", camera = {height = 6}}

local row = scene:group {}
for i = 1, 5 do row:rectangle {size = {0.8, 0.8}, corner = 0.1, fill = "accent"} end
row:arrange({1, 0}, 0.3)          -- left to right
row:move_to({-1, 1.6})

local column = scene:group {}
for i = 1, 4 do column:circle {radius = 0.3, fill = "warning"} end
column:arrange({0, -1}, 0.2)      -- top to bottom
column:move_to({3.5, 0})

scene:fade_in(row, {shift = {0, -0.3}, duration = 0.6})
scene:fade_in(column, {shift = {0.3, 0}, duration = 0.6})
scene:wait(0.5)

return scene
```

```js live
const scene = vis.scene({theme: "pro_black", camera: {height: 6}});

const row = scene.group({});
for (let i = 1; i <= 5; i++) row.rectangle({size: [0.8, 0.8], corner: 0.1, fill: "accent"});
row.arrange([1, 0], 0.3);          // left to right
row.moveTo([-1, 1.6]);

const column = scene.group({});
for (let i = 1; i <= 4; i++) column.circle({radius: 0.3, fill: "warning"});
column.arrange([0, -1], 0.2);      // top to bottom
column.moveTo([3.5, 0]);

scene.fadeIn(row, {shift: [0, -0.3], duration: 0.6});
scene.fadeIn(column, {shift: [0.3, 0], duration: 0.6});
scene.wait(0.5);

return scene;
```

### Grids

`arrange_grid(columns, col_gap, row_gap)` fills rows of `columns` children, top to bottom.

```lua live
local scene = vis.scene {theme = "pro_black", camera = {height = 6}}

local grid = scene:group {}
for i = 1, 12 do grid:rectangle {size = {0.7, 0.7}, corner = 0.1, fill = (i % 5 == 0) and "warning" or "secondary"} end
grid:arrange_grid(4, 0.25, 0.25)   -- 4 columns, then rows
grid:move_to({0, 0})

scene:fade_in(grid, {scale = 0.8, duration = 0.6})
scene:wait(0.5)

return scene
```

```js live
const scene = vis.scene({theme: "pro_black", camera: {height: 6}});

const grid = scene.group({});
for (let i = 1; i <= 12; i++) grid.rectangle({size: [0.7, 0.7], corner: 0.1, fill: i % 5 === 0 ? "warning" : "secondary"});
grid.arrangeGrid(4, 0.25, 0.25);   // 4 columns, then rows
grid.moveTo([0, 0]);

scene.fadeIn(grid, {scale: 0.8, duration: 0.6});
scene.wait(0.5);

return scene;
```

### next_to and align_to

`next_to(other, direction, gap)` puts an object beside another; `align_to(other, direction)` lines up one edge — here the red bar shares the box's top edge.

```lua live
local scene = vis.scene {theme = "pro_black", camera = {height = 6}}

local box = scene:rectangle {size = {2, 1.2}, corner = 0.1, fill = "accent"}
local tags = {}
for i, dir in ipairs({{1, 0}, {-1, 0}, {0, 1}, {0, -1}}) do
    tags[i] = scene:circle {radius = 0.25, fill = "warning"}
    tags[i]:next_to(box, dir, 0.3)     -- beside box, 0.3 apart
end
local bar = scene:rectangle {size = {0.6, 0.3}, fill = "danger"}
bar:move_to({3.5, 0})
bar:align_to(box, {0, 1})              -- same top edge as box

scene:grow_from_center(box, 0.4)
for _, tag in ipairs(tags) do scene:grow_from_center(tag, 0.25) end
scene:grow_from_edge(bar, "left", 0.4)
scene:wait(0.5)

return scene
```

```js live
const scene = vis.scene({theme: "pro_black", camera: {height: 6}});

const box = scene.rectangle({size: [2, 1.2], corner: 0.1, fill: "accent"});
const tags = [[1, 0], [-1, 0], [0, 1], [0, -1]].map((dir) =>
    scene.circle({radius: 0.25, fill: "warning"}).nextTo(box, dir, 0.3));   // beside box, 0.3 apart
const bar = scene.rectangle({size: [0.6, 0.3], fill: "danger"});
bar.moveTo([3.5, 0]);
bar.alignTo(box, [0, 1]);              // same top edge as box

scene.growFromCenter(box, 0.4);
for (const tag of tags) scene.growFromCenter(tag, 0.25);
scene.growFromEdge(bar, "left", 0.4);
scene.wait(0.5);

return scene;
```

### Animated rearrange

The timeline call `scene:arrange(items, duration)` moves siblings into a new row. Pass the list in its new order — here with a box inserted in the middle.

```lua live
local scene = vis.scene {theme = "pro_black", camera = {height = 6}}

local boxes = {}
for i = 1, 4 do
    boxes[i] = scene:rectangle {center = {-3 + i * 1.2, 0}, size = {0.8, 0.8}, corner = 0.1, fill = "accent"}
end
scene:wait(0.4)

-- insert a new box in the middle: re-arrange with it in the list
local new = scene:rectangle {center = {0, -1.5}, size = {0.8, 0.8}, corner = 0.1, fill = "warning"}
scene:fade_in(new, {duration = 0.3})
scene:arrange({boxes[1], boxes[2], new, boxes[3], boxes[4]}, 0.8, "smooth", {gap = 0.4})
scene:wait(0.6)

return scene
```

```js live
const scene = vis.scene({theme: "pro_black", camera: {height: 6}});

const boxes = [1, 2, 3, 4].map((i) =>
    scene.rectangle({center: [-3 + i * 1.2, 0], size: [0.8, 0.8], corner: 0.1, fill: "accent"}));
scene.wait(0.4);

// insert a new box in the middle: re-arrange with it in the list
const added = scene.rectangle({center: [0, -1.5], size: [0.8, 0.8], corner: 0.1, fill: "warning"});
scene.fadeIn(added, {duration: 0.3});
scene.arrange([boxes[0], boxes[1], added, boxes[2], boxes[3]], 0.8, "smooth", {gap: 0.4});
scene.wait(0.6);

return scene;
```

## Exits

<p>Every entrance has an exit.</p>
<table>
<tr><th>Call</th><th>Effect</th></tr>
<tr><td><code>uncreate</code></td><td>Erases a stroke in reverse.</td></tr>
<tr><td><code>fade_out</code></td><td>Fades out, optional <code>shift</code> / <code>scale</code>.</td></tr>
<tr><td><code>shrink_to_center</code></td><td>Collapses to its centre.</td></tr>
<tr><td><code>fade_transform(a, b)</code></td><td>Cross-fades; <code>b</code> replaces <code>a</code> (attach <code>b</code> just before the call).</td></tr>
<tr><td><code>transition(a, b)</code></td><td>Plain cross-fade of two independent handles.</td></tr>
<tr><td><code>remove(obj)</code></td><td>Drops it at the cursor.</td></tr>
</table>

### Leaving the stage

Four ways out, left to right: `uncreate` erases the stroke, `fade_out` fades with an optional drift, `shrink_to_center` collapses, and `remove` drops it at the cursor.

```lua live
local scene = vis.scene {theme = "pro_black", camera = {height = 6}}

local ring = scene:circle {center = {-4, 0}, radius = 0.7, stroke = "accent", width = 5}
local card = scene:rectangle {center = {-1.35, 0}, size = {1.4, 1.4}, corner = 0.15, fill = "success"}
local dot = scene:circle {center = {1.35, 0}, radius = 0.7, fill = "warning"}
local box = scene:rectangle {center = {4, 0}, size = {1.4, 1.4}, fill = "danger"}

scene:wait(0.4)
scene:uncreate(ring, 0.8)                                   -- erase the stroke
scene:fade_out(card, {shift = {0, -0.6}, duration = 0.6})   -- fade and drift
scene:shrink_to_center(dot, 0.6)                            -- collapse
scene:remove(box)                                           -- gone at once
scene:wait(0.6)

return scene
```

```js live
const scene = vis.scene({theme: "pro_black", camera: {height: 6}});

const ring = scene.circle({center: [-4, 0], radius: 0.7, stroke: "accent", width: 5});
const card = scene.rectangle({center: [-1.35, 0], size: [1.4, 1.4], corner: 0.15, fill: "success"});
const dot = scene.circle({center: [1.35, 0], radius: 0.7, fill: "warning"});
const box = scene.rectangle({center: [4, 0], size: [1.4, 1.4], fill: "danger"});

scene.wait(0.4);
scene.uncreate(ring, 0.8);                                  // erase the stroke
scene.fadeOut(card, {shift: [0, -0.6], duration: 0.6});     // fade and drift
scene.shrinkToCenter(dot, 0.6);                             // collapse
scene.remove(box);                                          // gone at once
scene.wait(0.6);

return scene;
```

### Replacing one object with another

`fade_transform(a, b)` hands `a`'s place to `b`; create `b` right before the call. To cross-fade two handles that stay independent, use `transition`.

```lua live
local scene = vis.scene {theme = "pro_black", camera = {height = 6}}

local before = scene:circle {fill = "accent"}
scene:wait(0.5)

local after = scene:rectangle {size = {2.4, 1.4}, corner = 0.2, fill = "warning"}
scene:fade_transform(before, after, 0.8)   -- after takes over from before
scene:wait(0.6)

return scene
```

```js live
const scene = vis.scene({theme: "pro_black", camera: {height: 6}});

const before = scene.circle({fill: "accent"});
scene.wait(0.5);

const after = scene.rectangle({size: [2.4, 1.4], corner: 0.2, fill: "warning"});
scene.fadeTransform(before, after, 0.8);   // after takes over from before
scene.wait(0.6);

return scene;
```

## Connectors and vectors

<p>A <code>connector</code> is derived geometry between two handles: it starts and ends on their edges and keeps following them as they
move. A <code>vector</code> is an arrow defined by <code>value</code> from an <code>origin</code>.</p>

### Connectors

Move either end and the connector follows: it is recomputed from the two handles every frame.

```lua live
local scene = vis.scene {theme = "pro_black", camera = {height = 6}}

local a = scene:circle {center = {-3, 1}, radius = 0.5, fill = "accent"}
local b = scene:rectangle {center = {3, -1}, size = {1.2, 0.8}, fill = "warning"}
local link = scene:connector {from = a, to = b, stroke = "#e6e8ee", width = 3, tip = 10, padding = 0.1}

scene:create(link, 0.6)
scene:parallel(                       -- the connector keeps both ends attached
    function() scene:shift(a, {1, -2.4}, 1.2, "smooth") end,
    function() scene:shift(b, {-1.5, 2.2}, 1.2, "smooth") end)

return scene
```

```js live
const scene = vis.scene({theme: "pro_black", camera: {height: 6}});

const a = scene.circle({center: [-3, 1], radius: 0.5, fill: "accent"});
const b = scene.rectangle({center: [3, -1], size: [1.2, 0.8], fill: "warning"});
const link = scene.connector(a, b, {stroke: "#e6e8ee", width: 3, tip: 10, padding: 0.1});

scene.create(link, 0.6);
scene.parallel(                       // the connector keeps both ends attached
    () => scene.shift(a, [1, -2.4], 1.2, "smooth"),
    () => scene.shift(b, [-1.5, 2.2], 1.2, "smooth"));

return scene;
```

### Vectors

Playing `value` swings the arrow to a new direction and length from the same origin.

```lua live
local scene = vis.scene {theme = "pro_black", camera = {height = 6}}

local v = scene:vector {origin = {0, 0}, value = {2, 1}, stroke = "danger", width = 4}

scene:create(v, 0.5)
scene:play({target = v, value = {-1, 2}}, 1)
scene:play({target = v, value = {2.5, -1.5}}, 1)

return scene
```

```js live
const scene = vis.scene({theme: "pro_black", camera: {height: 6}});

const v = scene.vector({origin: [0, 0], value: [2, 1], stroke: "danger", width: 4});

scene.create(v, 0.5);
scene.play({target: v, value: [-1, 2]}, 1);
scene.play({target: v, value: [2.5, -1.5]}, 1);

return scene;
```
